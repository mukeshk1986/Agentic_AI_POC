CREATE OR REPLACE VIEW ${catalog}.${schema_curation}.risk_member_output_blended AS
WITH risk_calculations AS (
  SELECT 
    RISK_MEMBER_ID,
    HOME_PLAN_ID_CD,
    MEMB_ID_CD,
    MEMBER_AGE,
    MEMBER_GENDER,
    ISSUER_ID,
    CONTRACT_ID,
    INCL_SUPP_MMR,
    INCL_PSEUDO_CLAIM,
    RISK_MODEL_TYPE,
    RISK_MODEL_VERSION,
    RISK_MODEL_YEAR,
    RISK_MODEL_SEGMENT,
    YEAR_MONTH,
    DEMOG_SCORE,
    DIAG_CD_TO_HCC_JSON_LIST,
    HCC_TO_SCORE_JSON_LIST,
    HCC_SCORE,
    HCC_COUNT_PAYMENT_SCORE,
    INTERACTION_SCORE,
    ORIGDS_SCORE,
    RISK_SCORE_RAW,
    RISK_SCORE_NORMALIZED,
    RISK_SCORE_PAYMENT_HCC,
    WEIGHTAGE,
    WEIGHTED_RISK_SCORE,
    DATE(CYCLE_RUN_DATE) AS RUN_DATE,
    CYCLE_RUN_DATE,
    ROUND(SUM(WEIGHTED_RISK_SCORE) OVER (
      PARTITION BY RISK_MEMBER_ID, RISK_MODEL_TYPE, RISK_MODEL_YEAR, 
                   YEAR_MONTH, DATE(CYCLE_RUN_DATE)
    ), 3) AS BLENDED_RISK_SCORE,
    COUNT(*) OVER (
      PARTITION BY RISK_MEMBER_ID, RISK_MODEL_TYPE, RISK_MODEL_YEAR, 
                   YEAR_MONTH, DATE(CYCLE_RUN_DATE)
    ) AS ROW_COUNT,
    CREATED_BY,
    CREATED_DATE 
  FROM ${catalog}.${schema_curation}.risk_member_output
)
SELECT 
  RISK_MEMBER_ID,
  HOME_PLAN_ID_CD,
  MEMB_ID_CD,
  MEMBER_AGE,
  MEMBER_GENDER,
  ISSUER_ID,
  CONTRACT_ID,
  INCL_SUPP_MMR,
  INCL_PSEUDO_CLAIM,
  RISK_MODEL_TYPE,
  RISK_MODEL_VERSION,
  RISK_MODEL_YEAR,
  RISK_MODEL_SEGMENT,
  YEAR_MONTH,
  DEMOG_SCORE,
  DIAG_CD_TO_HCC_JSON_LIST,
  HCC_TO_SCORE_JSON_LIST,
  HCC_SCORE,
  HCC_COUNT_PAYMENT_SCORE,
  INTERACTION_SCORE,
  ORIGDS_SCORE,
  RISK_SCORE_RAW,
  RISK_SCORE_NORMALIZED,
  RISK_SCORE_PAYMENT_HCC,
  WEIGHTAGE,
  WEIGHTED_RISK_SCORE,
  CASE 
    WHEN ROW_COUNT > 2 THEN WEIGHTED_RISK_SCORE 
    ELSE BLENDED_RISK_SCORE 
  END AS BLENDED_RISK_SCORE,
  ROW_COUNT,
  RUN_DATE,
  CYCLE_RUN_DATE,
  CREATED_BY,
  CREATED_DATE 
FROM risk_calculations;
